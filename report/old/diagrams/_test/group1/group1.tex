\newcommand{\CheckDiagram}[1]{%
  \sbox{\DiagramBox}{\input{../../diagram-#1.tex}\unskip}%
  \typeout{DIAGRAM-#1 WIDTH=\the\wd\DiagramBox, HEIGHT=\the\dimexpr\ht\DiagramBox+\dp\DiagramBox\relax}%
  \ifdim\wd\DiagramBox>168mm
    \typeout{DIAGRAM-#1 OVER WIDTH}%
  \fi
  \ifdim\dimexpr\ht\DiagramBox+\dp\DiagramBox\relax>180mm
    \typeout{DIAGRAM-#1 OVER HEIGHT}%
  \fi
  \null\vfill
  \noindent\makebox[\textwidth][c]{\usebox{\DiagramBox}}\par
  \vfill\null
  \clearpage
}
